#!/usr/bin/env python3
"""Rename to .py, place in an extracted Atlas folder and run after stopping Atlas.
Or use: python Atlas-Script-Names-Fix.py --folder "C:\\Atlas" "C:\\AtlasDemo"
Updates script discovery and deployment validation, preserving other files.
"""
import argparse,os,uuid
from pathlib import Path

OLD="{'Install':'_i','Uninstall':'_u','Repair':'_r'}"
NEW="{'Install':('_i','_install','_installs'),'Uninstall':('_u','_uninstall','_uninstalls'),'Repair':('_r','_repair','_repairs')}"
CAT_OLD="[('Install','_i'),('Uninstall','_u'),('Repair','_r')]"
CAT_NEW="[('Install',('_i','_install','_installs')),('Uninstall',('_u','_uninstall','_uninstalls')),('Repair',('_r','_repair','_repairs'))]"

def prepare(folder):
    folder=Path(folder).resolve()
    if not (folder/'package_catalog.py').is_file():
        raise ValueError('Choose the extracted Atlas folder containing package_catalog.py: '+str(folder))
    rules=[('package_catalog.py',CAT_OLD,CAT_NEW,1)]
    if (folder/'backend/main.py').is_file():rules.append(('backend/main.py',OLD,NEW,2))
    elif not (folder/'demo_server.py').is_file():raise ValueError('Unrecognised Atlas installation: '+str(folder))
    changes=[]
    for name,old,new,count in rules:
        path=folder/name
        if path.is_symlink():raise ValueError('Refusing linked source file: '+str(path))
        before=path.read_bytes();text=before.decode('utf-8')
        if text.count(old)==0 and text.count(new)==count:
            print('Already fixed:',path);continue
        if text.count(old)!=count or new in text:
            raise ValueError('Unexpected source version; no changes made: '+str(path))
        after=text.replace(old,new).encode('utf-8');compile(after,str(path),'exec')
        changes.append((path,before,after))
    return changes

def apply(folders):
    changes=[]
    for folder in dict.fromkeys(str(Path(f).resolve()) for f in folders):changes.extend(prepare(folder))
    token=uuid.uuid4().hex[:12];written=[]
    try:
        for path,before,after in changes:
            if path.read_bytes()!=before:raise RuntimeError('Source changed while patching: '+str(path))
            backup=path.with_name(path.name+'.before-script-names-'+token+'.bak')
            with backup.open('xb') as f:f.write(before)
            temp=path.with_name(path.name+'.'+token+'.tmp')
            try:
                with temp.open('xb') as f:f.write(after)
                os.replace(temp,path);written.append((path,before));print('Updated:',path,'Backup:',backup)
            finally:temp.unlink(missing_ok=True)
    except Exception:
        for path,before in reversed(written):path.write_bytes(before)
        raise
    print('Done. Restart Atlas/demo to refresh script lists. No database, package scripts or BAT settings changed.')

def main():
    p=argparse.ArgumentParser(description=__doc__)
    p.add_argument('--folder',type=Path,nargs='+',default=[Path(__file__).resolve().parent])
    a=p.parse_args();apply(a.folder)

if __name__=='__main__':
    try:main()
    except Exception as exc:raise SystemExit('PATCH FAILED: '+str(exc))
