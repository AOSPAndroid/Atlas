"""Put this file and the three Atlas-Part-N.txt files in one folder.
Rename this file Atlas-Join.py and run it with Python.
It rebuilds Atlas-Installer.py. Run that file to install Atlas.
No application or remote installation is executed by this joiner.
"""
from pathlib import Path
import hashlib,os,tempfile
PARTS = [('Atlas-Part-1.txt', 14000000, '2550c7a93cbcf7637c760fff9bfb9475ef40661e2804d8fb57177306f984492f'), ('Atlas-Part-2.txt', 14000000, 'f75e969a6c315ce8e17bce9e3337353076f207026097f728f411f78580956f2f'), ('Atlas-Part-3.txt', 12507709, '51dccdc961ae32306075f048c31961770cacad824dac9f8c215a9134aa61f0f8')]
EXPECTED = '82da5644a92421a7ea976ae2b8da53ad181da75b5480bcca8f9a0eb031ef6932'

def main():
    base=Path(__file__).resolve().parent
    target=base/'Atlas-Installer.py'
    if target.exists():
        if hashlib.sha256(target.read_bytes()).hexdigest()==EXPECTED:
            print('Atlas-Installer.py is already complete. Run it to install Atlas.')
            return
        raise ValueError('Atlas-Installer.py already exists with different content. Move or rename it first.')
    # Validate every part before writing the rebuilt installer.
    for name,size,digest in PARTS:
        path=base/name
        if not path.is_file():
            raise ValueError('Missing '+name+'. Put all three parts in this folder without renaming them.')
        content=path.read_bytes()
        if len(content)!=size or hashlib.sha256(content).hexdigest()!=digest:
            raise ValueError(name+' is incomplete or changed. Download that part again.')
    fd,temp=tempfile.mkstemp(prefix='atlas-join-',suffix='.tmp',dir=base)
    try:
        check=hashlib.sha256()
        with os.fdopen(fd,'wb') as writer:
            for name,_,_ in PARTS:
                content=(base/name).read_bytes();writer.write(content);check.update(content)
        if check.hexdigest()!=EXPECTED:raise ValueError('Final integrity check failed.')
        # Exclusive creation prevents overwriting another installer.
        with target.open('xb') as writer,open(temp,'rb') as reader:
            while block:=reader.read(1024*1024):writer.write(block)
        print('Success: Atlas-Installer.py rebuilt and verified. Run it to install Atlas.')
    finally:
        Path(temp).unlink(missing_ok=True)

if __name__=='__main__':
    try:main()
    except Exception as error:
        print('Could not join Atlas:',error)
        if os.name=='nt':input('Press Enter to close...')
        raise SystemExit(1)
    if os.name=='nt':input('Press Enter to close...')
